from datetime import date
import gspread
from oauth2client.service_account import ServiceAccountCredentials
import pandas as pd
import streamlit as st

# 1. ตั้งค่าการเชื่อมต่อ Google Sheets
@st.cache_resource
def init_connection():
  # ดึงข้อมูล Credential จาก Streamlit Secrets
  scope = [
      "https://spreadsheets.google.com/feeds",
      "https://www.googleapis.com/auth/drive",
  ]
  creds_dict = dict(st.secrets["gcp_service_account"])
  creds = ServiceAccountCredentials.from_json_keyfile_dict(creds_dict, scope)
  client = gspread.authorize(creds)
  return client


client = init_connection()

# เปิด Google Sheet (เปลี่ยนชื่อไฟล์ตามที่คุณตั้ง)
SHEET_NAME = "Trading Data Log"
sheet = client.open(SHEET_NAME).sheet1

st.title("🌱 Trading Data Input App")
st.markdown("ระบบบันทึกข้อมูลผลิตภัณฑ์และติดตามสถานะการส่งออก")

# 2. สร้างฟอร์มรับข้อมูล (Input Form) ทั้ง 12 ฟิลด์
with st.form("trading_form"):
  st.subheader("กรอกข้อมูลผลิตภัณฑ์")

  col1, col2 = st.columns(2)
  with col1:
    cultivar_name = st.text_input("1. Cultivar Name")
    supplier_name = st.text_input("2. Supplier Name")
    supplier_lot = st.text_input("3. Supplier Lot. No.")
    export_name = st.text_input("4. Export Name")
    thai_stick_lot = st.text_input("5. THAI STICK Lot. No.")
    grade = st.selectbox("6. Grade", ["Grade A", "Grade B", "Grade C", "Premium"])

  with col2:
    quantity = st.number_input("7. Quantity", min_value=0.0, step=0.1)
    inspection_req = st.selectbox("8. Inspection Req.", ["Yes", "No"])
    ir_req = st.selectbox("9. IR Req.", ["Yes", "No"])
    shipment_date = st.date_input("10. Shipment date", value=date.today())
    prod_type = st.text_input("11. Type")
    status = st.selectbox("12. Status", ["Pending", "In Progress", "Completed", "Cancelled"])

  submitted = st.form_submit_button("บันทึกข้อมูล (Save to Google Sheets)")

  if submitted:
    # เตรียมแถวข้อมูลเพื่อบันทึก
    row_data = [
        cultivar_name,
        supplier_name,
        supplier_lot,
        export_name,
        thai_stick_lot,
        grade,
        str(quantity),
        inspection_req,
        ir_req,
        str(shipment_date),
        prod_type,
        status,
    ]
    # Append ข้อมูลลง Google Sheets แถวถัดไป
    sheet.append_row(row_data)
    st.success("บันทึกข้อมูลลง Google Sheets เรียบร้อยแล้วครับ! 🎉")

# 3. แสดงข้อมูลที่มีอยู่ในปัจจุบันจาก Google Sheets
st.divider()
st.subheader("📋 ข้อมูล Trading ทั้งหมดในระบบ")
data = sheet.get_all_records()
if data:
  df = pd.DataFrame(data)
  st.dataframe(df, use_container_width=True)
else:
  st.info("ยังไม่มีข้อมูลในระบบ")
